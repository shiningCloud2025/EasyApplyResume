# EasyApplyResume - 轻量级推荐系统方案

## 一、设计目标

### 1.1 核心原则：不用大数据组件，纯 Java + MySQL

| 要求 | 实现方案 |
|------|---------|
| ❌ 不用 Spark/Flink | ✅ 用 Java Stream + SQL |
| ❌ 不用 Redis Cluster | ✅ 单机 Redis 缓存（可选） |
| ❌ 不用 Elasticsearch | ✅ MySQL 全文索引 |
| ❌ 不用复杂 ML 模型 | ✅ 简单算法 + 规则 |

### 1.2 推荐场景

| 场景 | 推荐内容 | 算法 |
|------|---------|------|
| 首页 | 推荐岗位 | 基于用户简历的内容匹配 |
| 模板页 | 推荐模板 | 基于行业 + 热度 |
| 岗位详情 | 相似岗位 | 基于内容相似度 |
| 求职攻略 | 相关文章 | 基于标签 + 阅读历史 |

### 4.3 基于内容的推荐

**ContentBasedStrategy.java**
```java
package com.zyh.easyapplyresume.recommend.strategy;

import com.hankcs.hanlp.HanLP;
import com.zyh.easyapplyresume.recommend.mapper.ItemTagMapper;
import com.zyh.easyapplyresume.recommend.mapper.UserBehaviorMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.*;
import java.util.stream.Collectors;

/**
 * 基于内容的推荐策略
 * 核心思想：推荐与用户历史浏览/收藏相似的物品
 */
@Component
public class ContentBasedStrategy implements RecommendStrategy {

    @Autowired
    private UserBehaviorMapper behaviorMapper;
    
    @Autowired
    private ItemTagMapper itemTagMapper;

    @Override
    public List<Integer> recommend(Integer userId, String itemType, int limit) {
        if (userId == null) {
            return Collections.emptyList(); // 未登录用户无法使用此策略
        }
        
        // 1. 获取用户历史行为的物品
        List<Integer> historyItemIds = behaviorMapper.selectUserHistoryItems(userId, itemType, 50);
        if (historyItemIds.isEmpty()) {
            return Collections.emptyList();
        }
        
        // 2. 提取用户偏好标签（从历史物品的标签中统计）
        Map<String, Double> userTags = extractUserTags(itemType, historyItemIds);
        
        // 3. 获取候选物品（排除已浏览的）
        List<Integer> candidateIds = itemTagMapper.selectCandidateItems(itemType, historyItemIds, limit * 3);
        
        // 4. 计算每个候选物品与用户偏好的相似度
        Map<Integer, Double> scores = new HashMap<>();
        for (Integer itemId : candidateIds) {
            double score = calculateSimilarity(userTags, itemType, itemId);
            scores.put(itemId, score);
        }
        
        // 5. 按相似度排序返回
        return scores.entrySet().stream()
            .sorted(Map.Entry.<Integer, Double>comparingByValue().reversed())
            .limit(limit)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }

    /**
     * 提取用户偏好标签
     */
    private Map<String, Double> extractUserTags(String itemType, List<Integer> itemIds) {
        Map<String, Double> tagWeights = new HashMap<>();
        
        // 从用户历史物品中提取所有标签
        List<Map<String, Object>> tags = itemTagMapper.selectTagsByItems(itemType, itemIds);
        
        for (Map<String, Object> tagInfo : tags) {
            String tag = (String) tagInfo.get("tag");
            Double weight = (Double) tagInfo.get("weight");
            tagWeights.merge(tag, weight, Double::sum);
        }
        
        // 归一化
        double total = tagWeights.values().stream().mapToDouble(Double::doubleValue).sum();
        if (total > 0) {
            tagWeights.replaceAll((k, v) -> v / total);
        }
        
        return tagWeights;
    }

    /**
     * 计算物品与用户偏好的相似度
     */
    private double calculateSimilarity(Map<String, Double> userTags, String itemType, Integer itemId) {
        List<Map<String, Object>> itemTags = itemTagMapper.selectTagsByItem(itemType, itemId);
        
        double score = 0.0;
        for (Map<String, Object> tagInfo : itemTags) {
            String tag = (String) tagInfo.get("tag");
            Double weight = (Double) tagInfo.get("weight");
            
            if (userTags.containsKey(tag)) {
                score += userTags.get(tag) * weight;
            }
        }
        
        return score;
    }

    @Override
    public String getName() {
        return "content_based";
    }
    
    @Override
    public double getWeight() {
        return 0.4; // 在混合推荐中的权重
    }
}
```

### 4.4 简化版协同过滤

**CollaborativeStrategy.java**
```java
package com.zyh.easyapplyresume.recommend.strategy;

import com.zyh.easyapplyresume.recommend.mapper.UserBehaviorMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.*;
import java.util.stream.Collectors;

/**
 * 简化版协同过滤推荐
 * 核心思想：找到相似用户，推荐他们喜欢但当前用户未看过的物品
 * 
 * 简化点：
 * 1. 不用矩阵分解，用简单的 Jaccard 相似度
 * 2. 不用全量用户，只找最近活跃的用户
 * 3. 用 SQL 直接计算，不用 Spark
 */
@Component
public class CollaborativeStrategy implements RecommendStrategy {

    @Autowired
    private UserBehaviorMapper behaviorMapper;

    @Override
    public List<Integer> recommend(Integer userId, String itemType, int limit) {
        if (userId == null) {
            return Collections.emptyList();
        }
        
        // 1. 获取当前用户的行为物品集合
        Set<Integer> userItems = new HashSet<>(
            behaviorMapper.selectUserHistoryItems(userId, itemType, 100)
        );
        
        if (userItems.isEmpty()) {
            return Collections.emptyList();
        }
        
        // 2. 找相似用户（有共同行为的用户）
        List<Integer> similarUsers = behaviorMapper.selectSimilarUsers(userId, itemType, userItems, 20);
        
        // 3. 获取相似用户喜欢但当前用户未看过的物品
        Map<Integer, Double> candidateScores = new HashMap<>();
        
        for (Integer similarUserId : similarUsers) {
            // 计算用户相似度（Jaccard）
            Set<Integer> otherItems = new HashSet<>(
                behaviorMapper.selectUserHistoryItems(similarUserId, itemType, 100)
            );
            double similarity = jaccardSimilarity(userItems, otherItems);
            
            // 相似用户看过但当前用户没看过的物品
            for (Integer itemId : otherItems) {
                if (!userItems.contains(itemId)) {
                    candidateScores.merge(itemId, similarity, Double::sum);
                }
            }
        }
        
        // 4. 按得分排序返回
        return candidateScores.entrySet().stream()
            .sorted(Map.Entry.<Integer, Double>comparingByValue().reversed())
            .limit(limit)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }

    /**
     * Jaccard 相似度
     */
    private double jaccardSimilarity(Set<Integer> set1, Set<Integer> set2) {
        Set<Integer> intersection = new HashSet<>(set1);
        intersection.retainAll(set2);
        
        Set<Integer> union = new HashSet<>(set1);
        union.addAll(set2);
        
        if (union.isEmpty()) {
            return 0.0;
        }
        return (double) intersection.size() / union.size();
    }

    @Override
    public String getName() {
        return "collaborative";
    }
    
    @Override
    public double getWeight() {
        return 0.3;
    }
}
```

### 4.6 混合推荐策略

**HybridStrategy.java**
```java
package com.zyh.easyapplyresume.recommend.strategy;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.*;
import java.util.stream.Collectors;

/**
 * 混合推荐策略
 * 综合多种策略的结果
 */
@Component
public class HybridStrategy implements RecommendStrategy {

    @Autowired
    private ContentBasedStrategy contentBasedStrategy;
    
    @Autowired
    private CollaborativeStrategy collaborativeStrategy;
    
    @Autowired
    private HotRankStrategy hotRankStrategy;

    @Override
    public List<Integer> recommend(Integer userId, String itemType, int limit) {
        Map<Integer, Double> scores = new HashMap<>();
        
        // 1. 基于内容推荐
        List<Integer> contentItems = contentBasedStrategy.recommend(userId, itemType, limit);
        addScores(scores, contentItems, contentBasedStrategy.getWeight());
        
        // 2. 协同过滤推荐
        List<Integer> cfItems = collaborativeStrategy.recommend(userId, itemType, limit);
        addScores(scores, cfItems, collaborativeStrategy.getWeight());
        
        // 3. 热度推荐（保底）
        List<Integer> hotItems = hotRankStrategy.recommend(userId, itemType, limit);
        addScores(scores, hotItems, hotRankStrategy.getWeight());
        
        // 4. 综合排序
        List<Integer> result = scores.entrySet().stream()
            .sorted(Map.Entry.<Integer, Double>comparingByValue().reversed())
            .limit(limit)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
        
        // 5. 如果结果不足，用热度补充
        if (result.size() < limit) {
            for (Integer itemId : hotItems) {
                if (!result.contains(itemId)) {
                    result.add(itemId);
                    if (result.size() >= limit) break;
                }
            }
        }
        
        return result;
    }

    private void addScores(Map<Integer, Double> scores, List<Integer> items, double weight) {
        int rank = 1;
        for (Integer itemId : items) {
            // 排名越靠前，分数越高
            double score = weight * (1.0 / Math.log(rank + 1));
            scores.merge(itemId, score, Double::sum);
            rank++;
        }
    }

    @Override
    public String getName() {
        return "hybrid";
    }
}
```

### 4.7 推荐服务

**RecommendServiceImpl.java**
```java
package com.zyh.easyapplyresume.recommend.service.impl;

import com.zyh.easyapplyresume.recommend.model.RecommendResult;
import com.zyh.easyapplyresume.recommend.service.RecommendService;
import com.zyh.easyapplyresume.recommend.strategy.HybridStrategy;
import com.zyh.easyapplyresume.recommend.strategy.HotRankStrategy;
import com.zyh.easyapplyresume.service.user.EmploymentInformationService;
import com.zyh.easyapplyresume.service.user.ResumeTemplateService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class RecommendServiceImpl implements RecommendService {

    @Autowired
    private HybridStrategy hybridStrategy;
    
    @Autowired
    private HotRankStrategy hotRankStrategy;
    
    @Autowired
    private EmploymentInformationService jobService;
    
    @Autowired
    private ResumeTemplateService templateService;

    @Override
    public RecommendResult<Object> recommendJobs(Integer userId, int limit) {
        List<Integer> itemIds;
        String strategy;
        
        if (userId != null) {
            // 登录用户：混合推荐
            itemIds = hybridStrategy.recommend(userId, "JOB", limit);
            strategy = "hybrid";
        } else {
            // 未登录用户：热度推荐
            itemIds = hotRankStrategy.recommend(null, "JOB", limit);
            strategy = "hot";
        }
        
        // 根据ID查询完整信息
        List<Object> jobs = jobService.selectByIds(itemIds);
        
        return new RecommendResult<>(jobs, strategy, itemIds.size());
    }

    @Override
    public RecommendResult<Object> recommendTemplates(Integer userId, Integer industryId, int limit) {
        List<Integer> itemIds;
        
        if (industryId != null) {
            // 有行业偏好：按行业筛选后再推荐
            itemIds = templateService.selectIdsByIndustry(industryId, limit);
        } else if (userId != null) {
            itemIds = hybridStrategy.recommend(userId, "TEMPLATE", limit);
        } else {
            itemIds = hotRankStrategy.recommend(null, "TEMPLATE", limit);
        }
        
        List<Object> templates = templateService.selectByIds(itemIds);
        return new RecommendResult<>(templates, "hybrid", itemIds.size());
    }

    @Override
    public RecommendResult<Object> recommendSimilarJobs(Integer jobId, int limit) {
        // 基于当前岗位的标签，找相似岗位
        // 简单实现：同行业 + 同城市 的其他岗位
        List<Integer> similarIds = jobService.selectSimilarJobIds(jobId, limit);
        List<Object> jobs = jobService.selectByIds(similarIds);
        return new RecommendResult<>(jobs, "similar", similarIds.size());
    }
}
```

### 4.8 用户行为采集服务

**UserBehaviorServiceImpl.java**
```java
package com.zyh.easyapplyresume.recommend.service.impl;

import com.zyh.easyapplyresume.recommend.mapper.UserBehaviorMapper;
import com.zyh.easyapplyresume.recommend.model.UserBehavior;
import com.zyh.easyapplyresume.recommend.service.UserBehaviorService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;

@Service
public class UserBehaviorServiceImpl implements UserBehaviorService {

    @Autowired
    private UserBehaviorMapper behaviorMapper;

    // 行为权重定义
    private static final double WEIGHT_VIEW = 1.0;
    private static final double WEIGHT_COLLECT = 3.0;
    private static final double WEIGHT_APPLY = 5.0;
    private static final double WEIGHT_SHARE = 2.0;

    /**
     * 记录用户行为（异步，不阻塞主流程）
     */
    @Async
    @Override
    public void recordBehavior(Integer userId, String targetType, Integer targetId, String behaviorType) {
        if (userId == null) return; // 未登录不记录
        
        UserBehavior behavior = new UserBehavior();
        behavior.setUserId(userId);
        behavior.setTargetType(targetType);
        behavior.setTargetId(targetId);
        behavior.setBehaviorType(behaviorType);
        behavior.setWeight(getWeight(behaviorType));
        behavior.setCreatedTime(LocalDateTime.now());
        
        // 防止重复记录（同一用户短时间内多次浏览同一物品）
        if ("VIEW".equals(behaviorType)) {
            boolean exists = behaviorMapper.existsRecent(userId, targetType, targetId, 5); // 5分钟内
            if (exists) return;
        }
        
        behaviorMapper.insert(behavior);
    }

    private double getWeight(String behaviorType) {
        switch (behaviorType) {
            case "VIEW": return WEIGHT_VIEW;
            case "COLLECT": return WEIGHT_COLLECT;
            case "APPLY": return WEIGHT_APPLY;
            case "SHARE": return WEIGHT_SHARE;
            default: return 1.0;
        }
    }

    @Override
    public void recordView(Integer userId, String targetType, Integer targetId) {
        recordBehavior(userId, targetType, targetId, "VIEW");
    }

    @Override
    public void recordCollect(Integer userId, String targetType, Integer targetId) {
        recordBehavior(userId, targetType, targetId, "COLLECT");
    }

    @Override
    public void recordApply(Integer userId, Integer jobId) {
        recordBehavior(userId, "JOB", jobId, "APPLY");
    }
}
```

**HotScoreTask.java**

```java
package com.zyh.easyapplyresume.recommend.task;

import com.zyh.easyapplyresume.recommend.mapper.ItemHotMapper;
import com.zyh.easyapplyresume.recommend.mapper.UserBehaviorMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Map;

/**
 * 热度分数计算定时任务
 * 每小时执行一次
 */
@Component
public class HotScoreTask {

    @Autowired
    private UserBehaviorMapper behaviorMapper;
    
    @Autowired
    private ItemHotMapper itemHotMapper;

    /**
     * 热度计算公式：
     * hot_score = view_count * 1 + collect_count * 3 + apply_count * 5
     * 
     * 时间衰减：最近7天的行为权重更高
     */
    @Scheduled(cron = "0 0 * * * ?") // 每小时执行
    public void calculateHotScore() {
        List<String> itemTypes = List.of("JOB", "TEMPLATE", "ADVICE");
        
        for (String itemType : itemTypes) {
            // 1. 统计最近7天的行为数据
            List<Map<String, Object>> stats = behaviorMapper.selectBehaviorStats(itemType, 7);
            
            for (Map<String, Object> stat : stats) {
                Integer itemId = (Integer) stat.get("item_id");
                Long viewCount = (Long) stat.getOrDefault("view_count", 0L);
                Long collectCount = (Long) stat.getOrDefault("collect_count", 0L);
                Long applyCount = (Long) stat.getOrDefault("apply_count", 0L);
                
                // 计算热度分数
                double hotScore = viewCount * 1.0 + collectCount * 3.0 + applyCount * 5.0;
                
                // 更新热度表
                itemHotMapper.upsertHotScore(itemType, itemId, 
                    viewCount.intValue(), collectCount.intValue(), applyCount.intValue(), hotScore);
            }
        }
    }
}
```

### 5.1 UserBehaviorMapper.xml

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" 
    "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.zyh.easyapplyresume.recommend.mapper.UserBehaviorMapper">

    <!-- 获取用户历史行为的物品ID -->
    <select id="selectUserHistoryItems" resultType="java.lang.Integer">
        SELECT DISTINCT target_id 
        FROM recommend_user_behavior
        WHERE user_id = #{userId} 
          AND target_type = #{targetType}
        ORDER BY created_time DESC
        LIMIT #{limit}
    </select>

    <!-- 找相似用户（有共同行为的用户） -->
    <select id="selectSimilarUsers" resultType="java.lang.Integer">
        SELECT user_id
        FROM recommend_user_behavior
        WHERE target_type = #{targetType}
          AND target_id IN 
            <foreach collection="userItems" item="id" open="(" separator="," close=")">
                #{id}
            </foreach>
          AND user_id != #{userId}
        GROUP BY user_id
        ORDER BY COUNT(*) DESC
        LIMIT #{limit}
    </select>

    <!-- 检查是否存在最近的相同记录 -->
    <select id="existsRecent" resultType="boolean">
        SELECT EXISTS (
            SELECT 1 FROM recommend_user_behavior
            WHERE user_id = #{userId}
              AND target_type = #{targetType}
              AND target_id = #{targetId}
              AND created_time > DATE_SUB(NOW(), INTERVAL #{minutes} MINUTE)
        )
    </select>

    <!-- 统计行为数据（用于热度计算） -->
    <select id="selectBehaviorStats" resultType="java.util.Map">
        SELECT 
            target_id as item_id,
            SUM(CASE WHEN behavior_type = 'VIEW' THEN 1 ELSE 0 END) as view_count,
            SUM(CASE WHEN behavior_type = 'COLLECT' THEN 1 ELSE 0 END) as collect_count,
            SUM(CASE WHEN behavior_type = 'APPLY' THEN 1 ELSE 0 END) as apply_count
        FROM recommend_user_behavior
        WHERE target_type = #{targetType}
          AND created_time > DATE_SUB(NOW(), INTERVAL #{days} DAY)
        GROUP BY target_id
    </select>

</mapper>
```
