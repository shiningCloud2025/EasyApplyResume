# EasyApplyResume - 国际化(i18n)实现方案

## 一、设计思路

### 1.1 核心原则：后端主导，前端最小改动

| 层级 | 职责 | 改动量 |
|------|------|--------|
| **后端** | 维护所有翻译文本、根据语言返回对应内容 | ⭐⭐⭐ 主要工作 |
| **前端** | 语言切换、传递语言参数、渲染后端返回的文本 | ⭐ 最小改动 |

### 1.2 整体架构

```
┌─────────────────┐     Accept-Language: zh-CN / en-US
│   前端 (React)   │ ──────────────────────────────────────►
│                 │                                          │
│  只需要:         │     ┌─────────────────────────────────┐  │
│  1. 语言切换按钮  │     │         后端 (Spring Boot)       │  │
│  2. 存储语言偏好  │     │                                 │  │
│  3. 渲染返回文本  │     │  ┌───────────────────────────┐  │  │
│                 │ ◄── │  │  I18nMessageSource        │  │  │
└─────────────────┘     │  │  (翻译文本管理)             │  │  │
                        │  └───────────────────────────┘  │  │
                        │                                 │  │
                        │  ┌───────────────────────────┐  │  │
                        │  │  LocaleResolver           │  │  │
                        │  │  (语言解析)               │  │  │
                        │  └───────────────────────────┘  │  │
                        └─────────────────────────────────┘
```

### 2.4 国际化配置类

**I18nConfig.java**
```java
package com.zyh.easyapplyresume.i18n.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.support.ResourceBundleMessageSource;
import org.springframework.web.servlet.LocaleResolver;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;
import org.springframework.web.servlet.i18n.AcceptHeaderLocaleResolver;
import org.springframework.web.servlet.i18n.LocaleChangeInterceptor;

import java.util.Arrays;
import java.util.Locale;

@Configuration
public class I18nConfig implements WebMvcConfigurer {

    /**
     * 支持的语言列表
     */
    public static final List<Locale> SUPPORTED_LOCALES = Arrays.asList(
        Locale.SIMPLIFIED_CHINESE,  // zh_CN
        Locale.US,                   // en_US
        Locale.JAPAN                 // ja_JP (可选)
    );

    /**
     * 消息资源配置
     */
    @Bean
    public ResourceBundleMessageSource messageSource() {
        ResourceBundleMessageSource source = new ResourceBundleMessageSource();
        source.setBasename("i18n/messages");
        source.setDefaultEncoding("UTF-8");
        source.setUseCodeAsDefaultMessage(true); // 找不到翻译时返回key本身
        return source;
    }

    /**
     * 语言解析器 - 从请求头 Accept-Language 获取
     */
    @Bean
    public LocaleResolver localeResolver() {
        AcceptHeaderLocaleResolver resolver = new AcceptHeaderLocaleResolver();
        resolver.setDefaultLocale(Locale.SIMPLIFIED_CHINESE); // 默认中文
        resolver.setSupportedLocales(SUPPORTED_LOCALES);
        return resolver;
    }

    /**
     * 语言切换拦截器 - 支持通过URL参数切换 ?lang=en_US
     */
    @Bean
    public LocaleChangeInterceptor localeChangeInterceptor() {
        LocaleChangeInterceptor interceptor = new LocaleChangeInterceptor();
        interceptor.setParamName("lang");
        return interceptor;
    }

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        registry.addInterceptor(localeChangeInterceptor());
    }
}
```

### 2.5 国际化服务

**I18nService.java**
```java
package com.zyh.easyapplyresume.i18n.service;

import java.util.Locale;
import java.util.Map;

public interface I18nService {
    
    /**
     * 获取单个翻译文本
     */
    String getMessage(String key);
    
    /**
     * 获取单个翻译文本（带参数）
     */
    String getMessage(String key, Object... args);
    
    /**
     * 获取指定语言的翻译
     */
    String getMessage(String key, Locale locale);
    
    /**
     * 批量获取翻译（按前缀）
     * 如 prefix="nav" 返回所有 nav.* 的翻译
     */
    Map<String, String> getMessagesByPrefix(String prefix);
    
    /**
     * 获取所有翻译文本（供前端一次性加载）
     */
    Map<String, String> getAllMessages();
    
    /**
     * 获取当前语言
     */
    Locale getCurrentLocale();
    
    /**
     * 获取支持的语言列表
     */
    List<Map<String, String>> getSupportedLocales();
}
```

**I18nServiceImpl.java**
```java
package com.zyh.easyapplyresume.i18n.service.impl;

import com.zyh.easyapplyresume.i18n.config.I18nConfig;
import com.zyh.easyapplyresume.i18n.service.I18nService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.MessageSource;
import org.springframework.context.i18n.LocaleContextHolder;
import org.springframework.core.io.Resource;
import org.springframework.core.io.ResourceLoader;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.util.*;

@Service
public class I18nServiceImpl implements I18nService {

    @Autowired
    private MessageSource messageSource;
    
    @Autowired
    private ResourceLoader resourceLoader;

    @Override
    public String getMessage(String key) {
        return getMessage(key, getCurrentLocale());
    }

    @Override
    public String getMessage(String key, Object... args) {
        return messageSource.getMessage(key, args, key, getCurrentLocale());
    }

    @Override
    public String getMessage(String key, Locale locale) {
        return messageSource.getMessage(key, null, key, locale);
    }

    @Override
    public Map<String, String> getMessagesByPrefix(String prefix) {
        Map<String, String> result = new HashMap<>();
        Map<String, String> all = getAllMessages();
        String prefixDot = prefix + ".";
        
        for (Map.Entry<String, String> entry : all.entrySet()) {
            if (entry.getKey().startsWith(prefixDot)) {
                // 返回不带前缀的key，方便前端使用
                String shortKey = entry.getKey().substring(prefixDot.length());
                result.put(shortKey, entry.getValue());
            }
        }
        return result;
    }

    @Override
    public Map<String, String> getAllMessages() {
        Map<String, String> messages = new LinkedHashMap<>();
        Locale locale = getCurrentLocale();
        
        try {
            // 构建资源文件路径
            String filename = "i18n/messages";
            if (!Locale.SIMPLIFIED_CHINESE.equals(locale)) {
                filename += "_" + locale.getLanguage() + "_" + locale.getCountry();
            }
            filename += ".properties";
            
            Resource resource = resourceLoader.getResource("classpath:" + filename);
            if (!resource.exists()) {
                // 回退到默认文件
                resource = resourceLoader.getResource("classpath:i18n/messages.properties");
            }
            
            Properties props = new Properties();
            props.load(new InputStreamReader(resource.getInputStream(), StandardCharsets.UTF_8));
            
            for (String key : props.stringPropertyNames()) {
                messages.put(key, props.getProperty(key));
            }
        } catch (IOException e) {
            // 加载失败返回空map
        }
        
        return messages;
    }

    @Override
    public Locale getCurrentLocale() {
        return LocaleContextHolder.getLocale();
    }

    @Override
    public List<Map<String, String>> getSupportedLocales() {
        List<Map<String, String>> list = new ArrayList<>();
        
        Map<String, String> zh = new HashMap<>();
        zh.put("code", "zh_CN");
        zh.put("name", "简体中文");
        zh.put("flag", "🇨🇳");
        list.add(zh);
        
        Map<String, String> en = new HashMap<>();
        en.put("code", "en_US");
        en.put("name", "English");
        en.put("flag", "🇺🇸");
        list.add(en);
        
        return list;
    }
}
```

### 2.6 国际化API控制器

**I18nController.java**
```java
package com.zyh.easyapplyresume.i18n.controller;

import com.zyh.easyapplyresume.bean.BaseResult;
import com.zyh.easyapplyresume.i18n.service.I18nService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/i18n")
@Tag(name = "国际化接口")
public class I18nController {

    @Autowired
    private I18nService i18nService;

    /**
     * 获取所有翻译文本（前端初始化时调用一次）
     */
    @Operation(summary = "获取所有翻译文本")
    @GetMapping("/messages")
    public BaseResult<Map<String, String>> getAllMessages() {
        return BaseResult.ok(i18nService.getAllMessages());
    }

    /**
     * 按前缀获取翻译（按需加载）
     */
    @Operation(summary = "按前缀获取翻译")
    @GetMapping("/messages/{prefix}")
    public BaseResult<Map<String, String>> getMessagesByPrefix(@PathVariable String prefix) {
        return BaseResult.ok(i18nService.getMessagesByPrefix(prefix));
    }

    /**
     * 获取支持的语言列表
     */
    @Operation(summary = "获取支持的语言列表")
    @GetMapping("/locales")
    public BaseResult<List<Map<String, String>>> getSupportedLocales() {
        return BaseResult.ok(i18nService.getSupportedLocales());
    }

    /**
     * 获取当前语言信息
     */
    @Operation(summary = "获取当前语言")
    @GetMapping("/current")
    public BaseResult<Map<String, Object>> getCurrentLocale() {
        Map<String, Object> result = new HashMap<>();
        result.put("locale", i18nService.getCurrentLocale().toString());
        result.put("language", i18nService.getCurrentLocale().getLanguage());
        result.put("country", i18nService.getCurrentLocale().getCountry());
        return BaseResult.ok(result);
    }
}
```

### 2.7 统一响应国际化

**修改 BaseResult.java** - 让错误信息自动翻译
```java
// 在现有 BaseResult 基础上添加静态方法

@Component
public class BaseResultHelper {
    
    private static I18nService i18nService;
    
    @Autowired
    public void setI18nService(I18nService service) {
        BaseResultHelper.i18nService = service;
    }
    
    /**
     * 返回国际化错误信息
     * @param messageKey 如 "error.unauthorized"
     */
    public static <T> BaseResult<T> error(String messageKey) {
        String message = i18nService != null 
            ? i18nService.getMessage(messageKey) 
            : messageKey;
        return BaseResult.error(message);
    }
    
    /**
     * 返回国际化错误信息（带参数）
     */
    public static <T> BaseResult<T> error(String messageKey, Object... args) {
        String message = i18nService != null 
            ? i18nService.getMessage(messageKey, args) 
            : messageKey;
        return BaseResult.error(message);
    }
}
```

### 3.2 国际化工具

**i18n/index.ts**
```typescript
import axios from 'axios'

// 翻译文本缓存
let messages: Record<string, string> = {}
let currentLocale = localStorage.getItem('locale') || 'zh_CN'

/**
 * 初始化国际化（应用启动时调用一次）
 */
export async function initI18n(): Promise<void> {
  try {
    const res = await axios.get('/api/i18n/messages', {
      headers: { 'Accept-Language': currentLocale.replace('_', '-') }
    })
    messages = res.data.data || {}
  } catch (e) {
    console.error('Failed to load i18n messages', e)
  }
}

/**
 * 获取翻译文本
 * @param key 翻译key，如 "nav.home"
 * @param defaultValue 默认值
 */
export function t(key: string, defaultValue?: string): string {
  return messages[key] || defaultValue || key
}

/**
 * 切换语言
 */
export async function setLocale(locale: string): Promise<void> {
  currentLocale = locale
  localStorage.setItem('locale', locale)
  await initI18n()
  // 刷新页面或触发重新渲染
  window.location.reload()
}

/**
 * 获取当前语言
 */
export function getLocale(): string {
  return currentLocale
}

export default { initI18n, t, setLocale, getLocale }
```

```tsx
  { code: 'en_US', name: 'English', flag: '🇺🇸' }
]

const LanguageSwitch: React.FC = () => {
  const currentLocale = getLocale()
  const current = languages.find(l => l.code === currentLocale) || languages[0]

  const items = languages.map(lang => ({
    key: lang.code,
    label: (
      <span>
        {lang.flag} {lang.name}
      </span>
    ),
    onClick: () => {
      if (lang.code !== currentLocale) {
        setLocale(lang.code)
      }
    }
  }))

  return (
    <Dropdown menu={{ items }} placement="bottomRight">
      <Button type="text" icon={<GlobalOutlined />}>
        {current.flag} {current.name}
      </Button>
    </Dropdown>
  )
}

export default LanguageSwitch
```

## 七、扩展：数据库存储翻译（可选）

如果需要动态管理翻译文本（如管理后台编辑），可以将翻译存入数据库：

```sql
CREATE TABLE `sys_i18n_message` (
    `id` INT PRIMARY KEY AUTO_INCREMENT,
    `message_key` VARCHAR(100) NOT NULL COMMENT '翻译key',
    `locale` VARCHAR(10) NOT NULL COMMENT '语言代码',
    `message` TEXT COMMENT '翻译文本',
    `created_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `updated_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY `uk_key_locale` (`message_key`, `locale`)
) COMMENT='国际化翻译表';
```

然后实现 `DatabaseMessageSource` 从数据库加载翻译。
