1.2项目结构
com.example.authdemo
├── config                # 配置类
│   ├── SecurityConfig.java        # Spring Security核心配置
│   └── RedisConfig.java           # Redis配置
├── security              # 认证授权核心组件
│   ├── authentication    # 认证相关
│   │   ├── MultiFactorLoginFilter.java  # 多因素登录过滤器
│   │   ├── MultiFactorToken.java        # 自定义认证Token
│   │   └── MultiFactorProvider.java     # 自定义认证提供者
│   ├── authorization     # 授权相关
│   │   └── MyAccessDecisionManager.java # 自定义授权决策管理器
│   ├── jwt               # JWT相关
│   │   ├── JwtAuthFilter.java           # JWT认证过滤器
│   │   └── JwtUtils.java                # JWT工具类
│   └── user              # 用户相关
│       └── LoginUser.java               # 自定义UserDetails实现
├── controller            # 控制器
│   ├── common            # 公共接口
│   │   └── VerifyCodeController.java    # 验证码接口
│   ├── user              # 用户接口
│   │   └── UserController.java          # 用户相关接口
│   └── admin             # 管理员接口
│       └── AdminController.java         # 管理员相关接口
├── service               # 服务层
│   ├── UserDetailsServiceImpl.java      # Spring Security用户服务
│   └── SysUserService.java              # 系统用户服务
├── mapper                # 数据访问层
│   └── SysUserMapper.java               # 用户Mapper
├── model                 # 数据模型
│   ├── entity            # 实体类
│   │   └── SysUser.java                 # 用户实体
│   └── vo                # 视图对象
│       └── LoginResponse.java           # 登录响应VO
├── util                  # 工具类
│   ├── RedisUtils.java                  # Redis工具类
│   ├── SecurityUtils.java               # 安全工具类
│   └── ResultUtil.java                  # 响应结果工具类
└── exception             # 异常处理
    └── GlobalExceptionHandler.java      # 全局异常处理器
